% Ambiente de Exemplo: Fluxo contínuo no papel branco
\newenvironment{exemplo}[1]{%
    \par\addvspace{1.2em}%
    \noindent\textbf{Exemplo #1.}\enspace\ignorespaces
}{%
    \penalty10000\hfill$\triangleleft$\par\addvspace{1.2em}%
}

\usepackage{tcolorbox}
